document.addEventListener('DOMContentLoaded', function () {
    // 1. Chart initialization
    const ctx = document.getElementById('revenueChart');
    if (ctx) {
        new Chart(ctx.getContext('2d'), {
            type: 'line',
            data: {
                labels: ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'],
                datasets: [{
                    label: 'Weekly Revenue ($)',
                    data: [1200, 1900, 1500, 2500, 2200, 3000, 4500],
                    borderColor: '#5d5fef',
                    backgroundColor: 'rgba(93, 95, 239, 0.1)',
                    borderWidth: 3,
                    fill: true,
                    tension: 0.4
                }]
            },
            options: {
                responsive: true,
                plugins: { legend: { display: false } },
                scales: {
                    y: { beginAtZero: true, grid: { color: 'rgba(255, 255, 255, 0.05)' }, ticks: { color: '#a0a0a0' } },
                    x: { grid: { display: false }, ticks: { color: '#a0a0a0' } }
                }
            }
        });
    }

    // 2. Control Panel Logic
    const runBtn = document.getElementById('runBtn');
    const consoleOutput = document.getElementById('consoleOutput');
    const scriptBox = document.getElementById('scriptContent');

    if (runBtn) {
        runBtn.addEventListener('click', function () {
            runBtn.disabled = true;
            runBtn.style.opacity = '0.5';
            runBtn.textContent = 'Processing...';
            consoleOutput.innerHTML += '<br>> [Anti-Gravity] AI 워크플로우를 시작합니다...';

            fetch('/run_generation', { method: 'POST' })
                .then(res => res.json())
                .then(data => {
                    if (data.status === 'success') {
                        consoleOutput.innerHTML += `<br>> [SUCCESS] 기획 및 제작 완료!`;
                        consoleOutput.innerHTML += `<br><pre style="color: #a0a0a0; font-size: 11px;">${data.output}</pre>`;
                        updateLatestAssets();
                    } else {
                        consoleOutput.innerHTML += `<br>> [ERROR] 오류 발생: ${data.message}`;
                    }
                    runBtn.disabled = false;
                    runBtn.style.opacity = '1';
                    runBtn.textContent = 'Start Automation Workflow';
                    consoleOutput.scrollTop = consoleOutput.scrollHeight;
                });
        });
    }

    function updateLatestAssets() {
        if (!scriptBox) return;
        fetch('/get_latest_assets')
            .then(res => res.json())
            .then(data => {
                if (data.script) {
                    scriptBox.innerHTML = `<strong>최신 대본:</strong><br>${data.script}`;
                }
                if (data.video_exists) {
                    consoleOutput.innerHTML += '<br>> [LOG] 최신 영상 파일(final_shorts.mp4)이 감지되었습니다.';
                }
            });
    }

    // 초기 상태 업데이트
    updateLatestAssets();
});
